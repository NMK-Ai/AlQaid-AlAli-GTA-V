cmake_minimum_required(VERSION 3.20)
project(OpenPilotGTA LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(BUILD_GTA_ADAPTER "Build the Windows ASI adapter (requires the Script Hook V SDK)" ${WIN32})
if(BUILD_GTA_ADAPTER)
  set(SCRIPTHOOK_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/ScriptHookV_SDK" CACHE PATH "Script Hook V SDK")
  if(NOT EXISTS "${SCRIPTHOOK_SDK}/inc/main.h")
    message(FATAL_ERROR "Obtain the official Script Hook V SDK first")
  endif()
  add_library(OpenPilotGTA SHARED bridge.cpp)
  target_include_directories(OpenPilotGTA PRIVATE "${SCRIPTHOOK_SDK}/inc")
  target_link_libraries(OpenPilotGTA PRIVATE "${SCRIPTHOOK_SDK}/lib/ScriptHookV.lib" ws2_32 xinput)
  target_compile_definitions(OpenPilotGTA PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  set_target_properties(OpenPilotGTA PROPERTIES PREFIX "" SUFFIX ".asi")
endif()
enable_testing()
add_executable(test_driving_state test_driving_state.cpp)
add_test(NAME driving_state COMMAND test_driving_state)
add_executable(test_angle_servo test_angle_servo.cpp)
add_test(NAME angle_servo COMMAND test_angle_servo)
add_executable(test_longitudinal_actuator test_longitudinal_actuator.cpp)
add_test(NAME longitudinal_actuator COMMAND test_longitudinal_actuator)
add_executable(test_radar test_radar.cpp)
add_test(NAME radar COMMAND test_radar)
add_executable(test_road_context test_road_context.cpp)
add_test(NAME road_context COMMAND test_road_context)

# These small contract tests use assert; keep checks active in Release builds.
foreach(test_target test_driving_state test_angle_servo test_longitudinal_actuator test_radar test_road_context)
  if(MSVC)
    target_compile_options(${test_target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${test_target} PRIVATE -UNDEBUG)
  endif()
endforeach()
